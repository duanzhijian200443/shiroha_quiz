cmake_minimum_required(VERSION 3.14)
project(shiroha_link_native LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Shiroha Link requires Windows x64")
endif()
foreach(role IN ITEMS 1 2 3)
  if(role EQUAL 1)
    set(target shiroha_link_native)
  elseif(role EQUAL 2)
    set(target shiroha_link_control_native)
  else()
    set(target shiroha_handoff_native)
  endif()
  add_library(${target} SHARED ../local_relay/local_relay.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE SHIROHA_LINK_ROLE=${role}
    UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /GS /guard:cf)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
  target_link_libraries(${target} PRIVATE advapi32 bcrypt user32)
endforeach()
